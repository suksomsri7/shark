type F = { attrs: string; fields: [string, string, string][] };
const unesc = (s: string) => s.replace(/&quot;/g, '"').replace(/&#x27;/g, "'").replace(/&#39;/g, "'").replace(/&lt;/g, "<").replace(/&gt;/g, ">").replace(/&amp;/g, "&");
function allForms(html: string): F[] {
  const out: F[] = [];
  for (const m of html.matchAll(/<form\b([^>]*)>([\s\S]*?)<\/form>/g)) {
    const fields: [string, string, string][] = [];
    for (const i of (m[2] ?? "").matchAll(/<input\b([^>]*)\/?>/g)) {
      const a = i[1] ?? "";
      const name = /\bname="([^"]*)"/.exec(a)?.[1];
      if (!name) continue;
      fields.push([unesc(name), /\btype="([^"]*)"/.exec(a)?.[1] ?? "text", unesc(/\bvalue="([^"]*)"/.exec(a)?.[1] ?? "")]);
    }
    out.push({ attrs: (m[1] ?? "").replace(/\s+/g, " ").slice(0, 200), fields });
  }
  return out;
}
/** ช่องอาร์กิวเมนต์ bind จริง: ค่าที่เป็น JSON array ยาว 3 ขึ้นต้นด้วย slug ร้าน */
const boundKey = (fields: [string, string][], slug: string): string | null => {
  for (const [k, v] of fields) {
    if (!/^\$ACTION_\d+:\d+$/.test(k)) continue;
    try { const j = JSON.parse(v) as unknown; if (Array.isArray(j) && j.length === 3 && j[0] === slug) return k; } catch { /* not json */ }
  }
  return null;
};
