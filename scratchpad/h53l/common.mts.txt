/** Hold a lock in its own tx; start each call only after the previous is observed blocked; then release. */
export async function underLock<R>(P: Any, lock: { key: string } | { rowSql: string; args: unknown[] }, calls: (() => Promise<R>)[]): Promise<{ res: R[]; blocked: boolean[] }> {
  let release: () => void = () => {};
  const gate = new Promise<void>((r) => (release = r));
  let held: (pid: number) => void = () => {};
  const heldP = new Promise<number>((r) => (held = r));
  const holder = P.$transaction(async (tx: Any) => {
    const pid = Number(((await tx.$queryRawUnsafe(`SELECT pg_backend_pid() AS p`)) as Any[])[0]?.p ?? -1);
    if ("key" in lock) await tx.$queryRawUnsafe(`SELECT pg_advisory_xact_lock(hashtextextended($1::text, 0))::text AS x`, lock.key);
    else await tx.$queryRawUnsafe(lock.rowSql, ...lock.args);
    held(pid);
    await gate;
  }, { maxWait: 20_000, timeout: 120_000 }).catch(() => held(-1));
  const pid = await heldP;
  const running: Promise<R>[] = [];
  const blocked: boolean[] = [];
  const nWaiting = async () => Number(((await P.$queryRawUnsafe(`SELECT count(*)::int AS n FROM pg_stat_activity WHERE datname = current_database() AND wait_event_type = 'Lock' AND cardinality(pg_blocking_pids(pid)) > 0`).catch(() => [{ n: 0 }])) as Any[])[0]?.n ?? 0);
  void pid;
  for (const call of calls) {
    const before = await nWaiting();
    let done = false;
    const p = call().finally(() => (done = true));
    running.push(p);
    let b = false;
    for (let i = 0; i < 80 && !done; i++) {
      await sleep(100);
      if ((await nWaiting()) > before) { b = true; break; }
    }
    blocked.push(b);
  }
  release();
  await holder;
  return { res: await Promise.all(running), blocked };
}
