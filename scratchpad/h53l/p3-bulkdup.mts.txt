// H5.3l hunter — bulk approve path with a duplicated request id (approval/actions.ts bulkDecideAction → bulkDecide, no de-dup)
// Run: bash scripts/iso.sh bash scripts/qc-hr.sh bash scripts/with-gate-lock.sh env QC_FORCE=1 pnpm exec tsx scratchpad/h53l/p3-bulkdup.mts
import { type Any, setup, short, errText, dist } from "./common.mjs";

const N = Number(process.env.H53L_N ?? 20);
const S = await setup("h53l-p3");
const { P, tag, TT, U, V, c, rows, safe, sys, fx } = S;
let crashed = "";
try {
  const AP = (await import("@/lib/modules/approval/service")) as Any;
  const SV = (await import("@/lib/modules/hr/service")) as Any;
  const LV = (await import("@/lib/modules/hr/leave")) as Any;
  const s = await sys("bulk");
  const pol = (await P.hrLeavePolicy.create({ data: { tenantId: TT, systemId: s, category: "VACATION", name: `พักร้อน ${tag}`, quotaDays: 300, paid: true } })).id as string;
  const e = await fx.mkEmp(TT, s, `พนง bulk ${tag}`, { startDate: "2030-01-01", unitId: S.u1 });
  await AP.createPolicy({ tenantId: TT }, { name: "สาย bulk", entityType: "HrLeave", systemId: s, steps: [{ order: 1, approverRole: "MANAGER" }, { order: 2, approverRole: "MANAGER" }] });
  const o: string[] = [];
  for (let i = 0; i < N; i++) {
    const d = new Date(Date.UTC(2031, 1, 2) + i * 86_400_000).toISOString().slice(0, 10);
    const lv = await SV.requestLeave(c(s), { employeeId: e, type: "VACATION", policyId: pol, fromDate: d, toDate: d, actorUserId: U.staff.id });
    const req = (await rows(`SELECT id FROM "ApprovalRequest" WHERE "entityId" = $1`, lv.id))[0].id as string;
    const r = await AP.bulkDecide(V.mgr, { tenantId: TT }, [req, req], "APPROVED", null);
    await safe(() => LV.applyLeaveDecision({ tenantId: TT, systemId: s, leaveId: lv.id, requestId: req, approved: true }));
    const st = (await rows(`SELECT r.status::text AS rs, l.status::text AS ls, (SELECT count(DISTINCT "decidedById")::int FROM "ApprovalDecision" WHERE "requestId" = r.id) AS who FROM "ApprovalRequest" r JOIN "HrLeave" l ON l.id = r."entityId" WHERE r.id = $1`, req))[0];
    o.push(`done=${r.done}·req=${st.rs}·leave=${st.ls}·distinctDeciders=${st.who}`);
  }
  console.log(`B1.bulkDuplicateId.MANAGER>MANAGER ${dist(o)}`);
} catch (x) {
  crashed = errText(x);
  console.log(`CRASH ${crashed} ${short((x as Error)?.stack?.split("\n").slice(1, 4))}`);
} finally {
  await S.cleanup();
}
process.exit(crashed ? 1 : 0);
