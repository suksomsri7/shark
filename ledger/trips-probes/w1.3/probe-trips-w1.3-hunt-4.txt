      } finally { await L.closeTab(t); }
    }
  });
} catch (e) {
  console.log(`FATAL ${String(e instanceof Error ? e.stack : e).slice(0, 800)}`);
} finally {
  await L.closeBrowser();
  await L.cleanup();
  L.summary("hunt-4");
}
