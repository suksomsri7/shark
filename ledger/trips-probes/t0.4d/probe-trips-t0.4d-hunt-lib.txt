export const KEY = "trips.public.unpaidHoldMaxPct";
export const shareLimit = (capacity: number, pct: number) => Math.ceil((capacity * pct) / 100);
export const BKK = (ymd: string, hm: string) => new Date(`${ymd}T${hm}:00+07:00`);
export const addMin = (d: Date | string, m: number) => new Date(new Date(d).getTime() + m * 60_000);
export const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));
export const hex = (n = 6) => randomBytes(n).toString("hex");
export const tok = () => randomBytes(24).toString("base64url");
export const sha256hex = (s: string) => createHash("sha256").update(s).digest("hex");
export const asRec = (v: unknown): Rec | null => (v !== null && typeof v === "object" && !Array.isArray(v) ? (v as Rec) : null);
export const R = (v: unknown): Rec => asRec(v) ?? {};
export const fmt = (v: unknown): string => { try { return (JSON.stringify(v) ?? String(v)).slice(0, 600); } catch { return String(v); } };
export const errMsg = (e: unknown) => (e instanceof Error ? e.message : String(e)).replace(/\s+/g, " ").slice(0, 300);
export const pct = (xs: number[], p: number) => { if (!xs.length) return 0; const s = [...xs].sort((a, b) => a - b); return s[Math.min(s.length - 1, Math.floor((p / 100) * s.length))]; };
export const stats = (xs: number[]) => `n=${xs.length} p50=${pct(xs, 50)}ms p95=${pct(xs, 95)}ms max=${xs.length ? Math.max(...xs) : 0}ms`;
export const tally = (rs: Rec[]) => { const m: Record<string, number> = {}; for (const r of rs) { const k = r.ok === true ? "ok" : `${String(r.code ?? r.ok)}${R(r.details).reason ? `:${String(R(r.details).reason)}` : ""}${r.e ? `:${String(r.e).slice(0, 60)}` : ""}`; m[k] = (m[k] ?? 0) + 1; } return m; };
export const BAD = (r: Rec) => r.ok === "THROW" || r.code === "BUSY" || /40P01|deadlock|P2028|P2034|P2024|55P03|Transaction already closed|Unable to start a transaction|Timed out fetching/i.test(`${r.e ?? ""} ${r.pg ?? ""} ${fmt(r.details ?? "")}`);

    else { found++; findings.push(name); console.log(`FOUND [${name}] expected ${expected} | got ${got}`); }
  };
  const note = (name: string, msg: string) => console.log(`NOTE [${name}] ${msg}`);
  const finish = async () => {
    try { await cleanupTenants(tenantIds); await cleanupUsers(); await rateCleanup(); } catch (e) { console.log(`  ⚠️ cleanup: ${errMsg(e)}`); }
    let left = -1;
    try { left = (await q<{ n: number }>(`SELECT (SELECT count(*) FROM "Tenant" WHERE slug LIKE $1) + (SELECT count(*) FROM "User" WHERE email LIKE $1) + (SELECT count(*) FROM "TripHold" WHERE "tenantId" = ANY($2::text[])) + (SELECT count(*) FROM "Party" WHERE "tenantId" = ANY($2::text[]))::int AS n`, [`${PFX}%`, tenantIds]))[0]?.n ?? -1; } catch (e) { console.log(`  ⚠️ leftover check: ${errMsg(e)}`); }
    console.log(`\n== ${label}: HELD ${held} · FOUND ${found}${findings.length ? ` (${findings.join(" | ")})` : ""} · leftover rows ${left}`);
    await pool.end().catch(() => undefined);
    const db = (await import("@/lib/core/db").catch(() => null)) as Rec | null;
    const pr = R(db?.prisma);
    if (typeof pr.$disconnect === "function") await (pr.$disconnect as () => Promise<void>)().catch(() => undefined);
  };

  const promoRule = async (params: Rec) => ins("TripPriceRule", { id: rid("ru"), tenantId: tA, unitId: uA, productId: null, kind: params.kind, code: params.code ?? null, params: JSON.stringify(params), priority: 0, active: true, usedCount: 0, updatedAt: new Date() });
  return { promoRule, STAMP, rid, q, pool, call, timed, tA, uA, userId, PRD, setV, mkDep, staff, sys, pub, withNow, line, booker, holdT, confirmInput, confirmT, bookNow, placeHoldPub, releaseBasket, closeDep, stopSell, sweep,
    depRow, publicHeld, channelHeld, publicGroups, ins, tenantIds, cleanupTenants, OWNER, SETM, TRM, inv, orphanParties, partyCount, sampler, withTimer, stuckTx, check, note, finish };
}
