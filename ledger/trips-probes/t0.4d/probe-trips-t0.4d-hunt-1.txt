import { addMin, BAD, fmt, hex, HOLD_MIN, huntSetup, KEY, NOW0, R, sleep, stats, tally, type Rec } from "./probe-trips-t0.4d-hunt-lib.mjs";

const H = await huntSetup("probe-t0.4d-hunt-1");
const { call, check, note, q } = H;
const rnd = (n: number) => Math.floor(Math.random() * n);
type Hd = Awaited<ReturnType<typeof H.holdT>>;
const ONLY = process.env.HUNT_ONLY ?? "abc";
let phN = 0;
const freshPhone = () => `09${String(10_000_000 + Number.parseInt(hex(3), 16) % 80_000_000 + phN++).slice(-8)}`;
const partiesByPhone = async (ph: string) => (await q<{ n: number }>(`SELECT count(*)::int n FROM "Party" WHERE "tenantId" = $1 AND "phoneNorm" = $2`, [H.tA, ph]))[0]?.n ?? -1;
const partiesByEmail = async (em: string) => (await q<{ n: number }>(`SELECT count(*)::int n FROM "Party" WHERE "tenantId" = $1 AND lower(trim(email)) = lower(trim($2))`, [H.tA, em]))[0]?.n ?? -1;
const partiesByName = async (nm: string) => (await q<{ n: number }>(`SELECT count(*)::int n FROM "Party" WHERE "tenantId" = $1 AND name = $2`, [H.tA, nm]))[0]?.n ?? -1;
const mkDeps = async (n: number, capacity: number) => { const out: string[] = []; for (let i = 0; i < n; i++) out.push(await H.mkDep({ key: ["A", "B", "C", "D"][i % 4], capacity })); return out; };
const quoted = async (ctx: Rec, lines: Rec[]) => { const qr = await call("quoteTrips", ctx, { lines }, {}); return R(qr.value); };
const bookNowQ = (ctx: Rec, lines: Rec[], quote: Rec, b: Rec, lat?: number[]) =>
  H.timed(call("bookTripsNow", ctx, { quote: { lines }, quoteId: quote.quoteId, quoteExpiresAt: quote.expiresAt, idempotencyKey: `n${hex(6)}`, booker: b, pay: { method: "PAY_LATER" } }, {}), lat);
const holdMany = async (n: number, mk: () => Promise<Rec[]>, ctx: () => Rec = () => H.pub()): Promise<Hd[]> => {
  const out: Hd[] = [];
  for (let i = 0; i < n; i += 10) out.push(...(await Promise.all(Array.from({ length: Math.min(10, n - i) }, async () => H.holdT(ctx(), await mk())))));
  const bad = out.filter((h) => h.r.ok !== true);
  if (bad.length) throw new Error(`setup holds failed ${bad.length}: ${fmt(bad[0].r)}`);
  return out;
};
const lockWaits = async () => (await q<{ n: number }>(`SELECT count(*)::int n FROM pg_stat_activity WHERE datname = current_database() AND wait_event_type = 'Lock'`))[0]?.n ?? 0;

try {
  await H.setV([{ key: KEY, scope: "UNIT", scopeId: H.uA, value: 100 }, { key: "trips.public.holdCapPerContact", scope: "UNIT", scopeId: H.uA, value: 20 }]);

  if (ONLY.includes("a")) {
    // a1 STAFF bookTripsNow N-way, same NEW phone, different departures
    for (const [n, rounds] of [[10, 5], [50, 2]] as const) {

      for (let r = 0; r < rounds; r++) {

        const rs = await H.withTimer(Promise.all(qs.map(({ l, qt }) => bookNowQ(H.staff(), [l], qt, b, lat))), 180_000, `a1 ${n}`);
        if (rs === "TIMEOUT") { bads.push({ ok: "TIMEOUT" }); continue; }
        okN += rs.filter((x) => x.ok === true).length; bads.push(...rs.filter(BAD));
        per.push(await partiesByPhone(ph));
      }
      check(`a1 STAFF bookTripsNow ${n}-way same new phone, ${n} departures ×${rounds} ⇒ exactly 1 Party each round · all ok · no BUSY/40P01`, per.every((x) => x === 1) && okN === n * rounds && bads.length === 0,
        `Party 1×${rounds}`, `Party per round ${fmt(per)} · ok ${okN}/${n * rounds} · bad ${bads.length} ${fmt(bads[0])} · ${stats(lat)}`);
    }
    // a2 one phone in 6 spellings, 12-way (STAFF bookNow + PUBLIC confirm mixed)
    {
      const per: number[] = []; const tl: Rec[] = [];

        const ph = freshPhone(); const d9 = ph.slice(1);
        const forms = [ph, `+66${d9}`, `${ph.slice(0, 3)}-${ph.slice(3, 6)}-${ph.slice(6)}`, `${ph.slice(0, 3)} ${ph.slice(3, 6)} ${ph.slice(6)}`, `+66 ${d9.slice(0, 2)} ${d9.slice(2, 5)} ${d9.slice(5)}`, `0066${d9}`];

        "1×4", `per round ${fmt(per)} · ${fmt(tally(tl))}`);

    }

      }

        const ds = await mkDeps(2, 10); const ph = freshPhone(); const em = `t04dh.${hex(4)}@example.com`; const nm = `t04dh a4s ${em}`;
        const qs = await Promise.all(ds.map(async (d) => { const l = await H.line(d, 1); return { l, qt: await quoted(H.staff(), [l]) }; }));
        if (r % 2) { await bookNowQ(H.staff(), [qs[0].l], qs[0].qt, { name: nm, phone: ph, email: em }); await bookNowQ(H.staff(), [qs[1].l], qs[1].qt, { name: nm, email: em }); }
        else { await bookNowQ(H.staff(), [qs[1].l], qs[1].qt, { name: nm, email: em }); await bookNowQ(H.staff(), [qs[0].l], qs[0].qt, { name: nm, phone: ph, email: em }); }

      ];

    for (const [op, rs] of Object.entries(byOp)) note(`b ${op}`, `${fmt(tally(rs))} · ${stats(lat[op] ?? [])}`);

  // ═════ (b2) reviewer (2): same booker ×10 confirming on D ∥ payments / change / refund requests on D's bookings · unrelated customers' wait ═════

  }

  // ═════ (c) cost ═════
  if (ONLY.includes("c")) {
    const conf = async (n: number, sameBooker: boolean, actor: "STAFF" | "PUBLIC") => {
      const d = await H.mkDep({ capacity: 400 });
      const hs = await holdMany(n, async () => [await H.line(d, 1)], actor === "STAFF" ? () => H.staff() : () => H.pub());
      const ph = freshPhone();
      const lat: number[] = []; const t0 = Date.now();
      const rs = await Promise.all(hs.map((h) => H.confirmT(h, { booker: sameBooker ? { name: `t04dh c same ${ph}`, phone: ph } : { name: `t04dh c ${hex(3)}`, phone: freshPhone() } }, lat)));
      const wall = Date.now() - t0;
      return { lat, wall, t: tally(rs), parties: sameBooker ? await partiesByPhone(ph) : -1 };
    };
    for (const n of [30, 60]) {
      for (let rep = 0; rep < 2; rep++) {
        const dI = await conf(n, false, "STAFF"); const sA = await conf(n, true, "STAFF");
        note(`c ${n} STAFF confirms on one departure (rep ${rep})`, `distinct bookers ${stats(dI.lat)} wall ${dI.wall}ms ${fmt(dI.t)} · same NEW booker ${stats(sA.lat)} wall ${sA.wall}ms ${fmt(sA.t)} Party ${sA.parties}`);
      }
    }
    for (let rep = 0; rep < 2; rep++) {
      const dI = await conf(20, false, "PUBLIC"); const sA = await conf(20, true, "PUBLIC");
      note(`c 20 PUBLIC confirms (contact cap 20) one departure (rep ${rep})`, `distinct ${stats(dI.lat)} wall ${dI.wall}ms · same booker ${stats(sA.lat)} wall ${sA.wall}ms ${fmt(sA.t)} Party ${sA.parties}`);
    }
    // spam: 20 other customers (PUBLIC) ∥ {nothing | 50 STAFF confirms same booker | 50 STAFF confirms distinct bookers | 50 PUBLIC confirms same phone}
    const others = async (noise: "none" | "staffSame" | "staffDistinct" | "pubSame") => {
      const d = await H.mkDep({ capacity: 400 });
      const oh = await holdMany(20, async () => [await H.line(d, 1)]);
      const nh = noise === "none" ? [] : await holdMany(50, async () => [await H.line(d, 1)], noise === "pubSame" ? () => H.pub() : () => H.staff());
      const ph = freshPhone();
      const olat: number[] = []; const nlat: number[] = [];
      const noiseP = nh.map((h, i) => H.confirmT(h, { booker: noise === "staffDistinct" ? { name: `t04dh n ${i}`, phone: freshPhone() } : { name: `t04dh spam ${ph}`, phone: ph } }, nlat));
      await sleep(noise === "none" ? 0 : 30);
      const ors = await Promise.all(oh.map((h) => H.confirmT(h, { booker: { name: `t04dh o ${hex(3)}`, phone: freshPhone() } }, olat)));
      const nrs = await Promise.all(noiseP);
      return `others ${stats(olat)} ${fmt(tally(ors))} · noise ${stats(nlat)} ${fmt(tally(nrs))}`;
    };
    for (let rep = 0; rep < 2; rep++) for (const m of ["none", "staffSame", "staffDistinct", "pubSame"] as const) note(`c spam ${m} (rep ${rep})`, await others(m));
  }
} catch (e) {
  console.log(`💥 probe crashed: ${String(e instanceof Error ? e.stack : e).slice(0, 800)}`);
} finally {
  await H.finish();
}
