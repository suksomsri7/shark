async function press(name: string, sel: string, opts: { tab: string; flash: string; before?: () => Promise<void>; nav?: boolean } ) {
  n += 1;
  const e0 = errors.length;
  try {
    if (opts.before) await opts.before();

    const nav = opts.nav === false ? null : page.waitForNavigation({ waitUntil: "networkidle2", timeout: 30_000 }).catch(() => null);
    await el.click();
    if (nav) await nav;
    await page.waitForNetworkIdle({ idleTime: 500, timeout: 15_000 }).catch(() => {});
    await sleep(600);
    const body = (await page.evaluate("document.body.innerText")) as string;
    const crash = CRASH.test(body);
    const url = page.url().replace(BASE, "");
    const tab = new URL(page.url()).searchParams.get("tab");
    const inPanel = opts.flash ? await page.$(`[data-testid="hr-set-panel-${opts.tab}"] ${opts.flash}`) : null;
    const flash = opts.flash ? await txt(`[data-testid="hr-set-panel-${opts.tab}"] ${opts.flash}`) : "";
    const png = `${OUT}/${String(n).padStart(2, "0")}-${name}.png`;
    await page.screenshot({ path: png, fullPage: false });
    results.push({ name, ok: !crash && (!opts.flash || !!inPanel), crash, url, tab, flash, errors: errors.slice(e0), png });
  } catch (e) {
    results.push({ name, ok: false, why: String((e as Error).message).slice(0, 200), errors: errors.slice(e0) });
  }
