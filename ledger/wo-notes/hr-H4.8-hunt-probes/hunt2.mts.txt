const AK = (await import("@/lib/api-keys/service")) as Any;

type Res = { status: number; body: Any; text: string; headers: Headers };
async function api(method: string, path: string, headers: Record<string, string>): Promise<Res> {
  const pathOnly = path.split("?")[0]!;
  try {
    const res: Response = await RT[method](new Request(`http://x/api/v1/hr${path}`, { method, headers }), { params: Promise.resolve({ path: pathOnly.split("/").filter(Boolean) }) });
    const text = await res.text();
    let body: Any = null;
    try {
      body = JSON.parse(text);
    } catch {
      body = { _raw: text };
    }
    return { status: res.status, body, text, headers: res.headers };
  } catch (e) {
    return { status: -1, body: null, text: errText(e), headers: new Headers() };
  }
}
const sc = (r: Res) => `${r.status}${r.body?.error?.code ? `:${r.body.error.code}` : ""}`;

  const k = await mk("r", ["hr.employee.read", "hr.roster.read", "hr.leave.read"]);
  const H = { authorization: `Bearer ${k}` };
  const d = await api("GET", "/attendance/day?day=2026-06-11", H);

  const d3 = await api("GET", "/attendance/day?day=0000-00-00", H);

async function cleanMine(): Promise<void> {
  const ids = (((await fx.rawRows(`SELECT "id" FROM "ApiKey" WHERE "tenantId" = ANY($1::text[])`, fx.tenants)) ?? []) as Any[]).map((r) => String(r.id));

  for (const t of ["ApiIdempotency", "ApiKey"]) await fx.rawRows(`DELETE FROM "${t}" WHERE "tenantId" = ANY($1::text[])`, fx.tenants);
}
