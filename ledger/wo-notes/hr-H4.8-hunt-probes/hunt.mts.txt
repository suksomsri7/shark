type Res = { status: number; body: Any; text: string; headers: Headers };
async function api(method: string, path: string, raw: string | null, o: { body?: unknown; idem?: string | false; headers?: Record<string, string>; rawBody?: string } = {}): Promise<Res> {
  const fn = RT[method];
  if (typeof fn !== "function") return { status: 0, body: null, text: "no-method", headers: new Headers() };
  const headers: Record<string, string> = { ...(raw ? { authorization: `Bearer ${raw}` } : {}), ...(o.headers ?? {}) };
  if (method !== "GET" && o.idem !== false) headers["idempotency-key"] = o.idem ?? `hunt-${tag}-${++n}`;
  const b = o.rawBody ?? (o.body === undefined ? undefined : JSON.stringify(o.body));
  if (b) headers["content-type"] = "application/json";
  try {
    const pathOnly = path.split("?")[0]!;
    const res: Response = await fn(new Request(`http://x/api/v1/hr${path}`, { method, headers, body: b }), { params: Promise.resolve({ path: pathOnly.split("/").filter(Boolean).map((s) => decodeURIComponent(s)) }) });
    const text = await res.text();
    let body: Any = null;
    try {
      body = JSON.parse(text);
    } catch {
      body = { _raw: text };
    }
    return { status: res.status, body, text, headers: res.headers };
  } catch (e) {
    return { status: -1, body: { error: { code: "THROW", message: errText(e) } }, text: errText(e), headers: new Headers() };
  }
}
