function unzipText(buf: Buffer): string {
  let eocd = -1;
  for (let i = buf.length - 22; i >= 0; i--) if (buf.readUInt32LE(i) === 0x06054b50) { eocd = i; break; }
  if (eocd < 0) return "";
  const n = buf.readUInt16LE(eocd + 10);
  let off = buf.readUInt32LE(eocd + 16);
  let out = "";
  for (let k = 0; k < n; k++) {
    if (buf.readUInt32LE(off) !== 0x02014b50) break;
    const method = buf.readUInt16LE(off + 10), csize = buf.readUInt32LE(off + 20), fnl = buf.readUInt16LE(off + 28), exl = buf.readUInt16LE(off + 30), cml = buf.readUInt16LE(off + 32), loc = buf.readUInt32LE(off + 42);
    const lfn = buf.readUInt16LE(loc + 26), lex = buf.readUInt16LE(loc + 28);
    const data = buf.subarray(loc + 30 + lfn + lex, loc + 30 + lfn + lex + csize);
    try { out += "\n" + (method === 8 ? inflateRawSync(data) : data).toString("utf8"); } catch { /* skip */ }
    off += 46 + fnl + exl + cml;
  }
  return out;
}
function pdfText(buf: Buffer): string {
  // raw + inflated streams (text is usually hex glyph ids, so sentinel search here is best-effort)
  let s = buf.toString("latin1");
  const re = /stream\r?\n/g;
  let m: RegExpExecArray | null;
  const parts: string[] = [s];
  while ((m = re.exec(s))) {
    const start = m.index + m[0].length;
    const end = s.indexOf("endstream", start);
    if (end < 0) break;
    try { parts.push(inflateSync(buf.subarray(start, end)).toString("latin1")); } catch { /* not flate */ }
  }
  return parts.join("\n");
}
function textOfAny(v: unknown): string {
  const seen = new WeakSet();
  const out: string[] = [];
  const walk = (x: unknown) => {
    if (x === null || x === undefined) return;
    if (x instanceof Uint8Array || Buffer.isBuffer(x)) {
      const b = Buffer.from(x as Uint8Array);
      if (b.subarray(0, 2).toString() === "PK") out.push(unzipText(b));
      else if (b.subarray(0, 4).toString() === "%PDF") out.push(pdfText(b));
      else out.push(b.toString("utf8"));
      return;
    }
    if (x instanceof Date) return void out.push(x.toISOString());
    if (typeof x === "object") {
      if (seen.has(x as object)) return;
      seen.add(x as object);
      for (const [k, y] of Object.entries(x as object)) { out.push(k); walk(y); }
      return;
    }
    out.push(String(x));
  };
  walk(v);
  return out.join("\u0001");
}

      if (base === "payU1") return homeU1; // H3.8 Q2: home unit only
      return false;
    case "CONTACT":
      if (["owner", "payAll", "hrAdmin"].includes(base)) return true;
      if (base === "payU1" || base === "mgrU1") return homeU1 || borrowedU1;
      return false;
    case "REASON":
    case "NOTE":
      if (["owner", "payAll", "hrAdmin"].includes(base)) return true; // lenient — triaged by hand
      if (base === "payU1" || base === "mgrU1") return homeU1 || borrowedU1;
      return false;
    case "CLOCK":
      if (["owner", "hrAdmin", "payAll"].includes(base)) return true;

const MATRIX: { surface: string; viewer: string; target: string; outcome: "data" | "refused" | "error" | "missing"; leaks: string[]; refusal?: string }[] = [];
function outcomeOf(r: Any): "data" | "refused" | "error" | "missing" {
  if (r && typeof r === "object" && "__missing" in r) return "missing";
  if (r && typeof r === "object" && "__throw" in r) return "error";
  if (r === null || r === undefined || r === false) return "refused";
  if (typeof r === "string") { try { const j = JSON.parse(r); if (j && typeof j === "object" && ("error" in j) && Object.keys(j).length <= 2) return "refused"; } catch { /* text */ } return "data"; }
  if (typeof r === "object" && r.ok === false) return "refused";
  if (Array.isArray(r) && r.length === 0) return "refused";
  if (typeof r === "object" && "status" in r && typeof r.status === "number" && r.status >= 400) return "refused";
  return "data";
}
function scan(F: Any, vk: VK, surface: string, r: Any): string[] {
  const hay = textOfAny(r);
  const hay2 = hay.replace(/[-\s]/g, "");
  const leaks: string[] = [];
  for (const s of SENT) {
    if (!s.s || s.s.length < 4) continue;
    const hit = hay.includes(s.s) || (s.field === "nationalId" && hay2.includes(s.s));
    if (hit && !allowed(F, vk, surface, s.emp, s.cls)) leaks.push(`${s.emp}.${s.field}`);
  }
  return [...new Set(leaks)];
}
function rec(F: Any, surface: string, vk: VK, target: string, r: Any) {
  const outcome = outcomeOf(r);
  const leaks = scan(F, vk, surface, r);
  MATRIX.push({ surface, viewer: vk, target, outcome, leaks, ...(outcome !== "data" ? { refusal: (typeof r === "string" ? r : canon(r)).slice(0, 300) } : {}) });
}
